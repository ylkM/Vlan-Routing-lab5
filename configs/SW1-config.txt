! ============================================
! SW1 Configuration
! Creates VLANs 10/20/30, assigns access ports
! ============================================

enable
configure terminal

hostname SW1

! --- Create and name the VLANs ---
vlan 10
 name Engineering
 exit

vlan 20
 name HR
 exit

vlan 30
 name Sales
 exit

! --- Engineering PCs ---
interface FastEthernet3/1
 description Eng-PC1 (10.0.0.1)
 switchport mode access
 switchport access vlan 10
 exit

interface FastEthernet4/1
 description Eng-PC2 (10.0.0.2)
 switchport mode access
 switchport access vlan 10
 exit

! --- HR PCs ---
interface FastEthernet5/1
 description HR-PC1 (10.0.0.65)
 switchport mode access
 switchport access vlan 20
 exit

interface FastEthernet6/1
 description HR-PC2 (10.0.0.66)
 switchport mode access
 switchport access vlan 20
 exit

! --- Sales PCs ---
interface FastEthernet7/1
 description Sales-PC1 (10.0.0.130)
 switchport mode access
 switchport access vlan 30
 exit

interface FastEthernet8/1
 description Sales-PC2 (10.0.0.129)
 switchport mode access
 switchport access vlan 30
 exit

! --- Links to R1 (each carries only its own VLAN, access mode, NOT trunk) ---
interface FastEthernet1/1
 description Link to R1 G0/0 -- VLAN10 gateway
 switchport mode access
 switchport access vlan 10
 exit

interface FastEthernet2/1
 description Link to R1 G0/1 -- VLAN20 gateway
 switchport mode access
 switchport access vlan 20
 exit

interface GigabitEthernet1/1
 description Link to R1 Gig0/2 -- VLAN30 gateway
 switchport mode access
 switchport access vlan 30
 exit

end
write memory

! ============================================
! Verification commands
! ============================================
! show vlan brief
! show interfaces status
